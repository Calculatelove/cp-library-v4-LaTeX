template<class T> struct Fenwick {
    vector<T> tree;
    explicit Fenwick(int n) : tree(n + 1, 0) {}
    void add(int i, T delta) {
        for (; i < (int)tree.size(); i += i & -i)
            tree[i] += delta;
    }
    T sum(int i) const {
        T result = 0;
        while (i > 0) { result += tree[i]; i -= i & -i; }
        return result;
    }
};
