% 在主文件的 \documentclass 之后加载；建议使用 XeLaTeX 编译。
% 此文件集中加载宏包；各版参数放在 layout-<版本>.tex，三版共用的排版实现放在 shared/，自定义命令放在 mystyle.tex。

% 中文支持；不采用 ctex 的中文标题名称与版式方案。
% 暂用 Fandol；字体选项必须在加载 ctex 前传入，最终字体方案尚未实现。
\PassOptionsToPackage{fontset=fandol}{ctex}
\usepackage[scheme=plain]{ctex}

% 数学公式、矩阵、对齐环境与数学符号（含 \mathbb）。
\usepackage{amsmath}
\usepackage{amssymb}
% OpenType 数学字体；各版在 layout 文件中选择具体字体。
\usepackage{unicode-math}
% 保留正文 Unicode 字符，由各版按需设置符号字体映射。
\usepackage{newunicodechar}

% 图片与颜色。
\usepackage{graphicx}
\usepackage{xcolor}

% 删除线；normalem 保留 \emph 的原有强调语义。
\usepackage[normalem]{ulem}

% 页面尺寸、边距与多栏排版；具体参数在各版 layout-<版本>.tex 中设置。
\usepackage{geometry}
\usepackage{multicol}
\usepackage{titlesec}
\usepackage{fancyhdr}
\usepackage{enumitem}
\usepackage{tocloft}

% 代码高亮，以及后续代码标题栏、引用块所需的盒子支持。
\usepackage{minted}
\usepackage{tcolorbox}
\tcbuselibrary{skins,breakable,minted}

% 目录链接、URL 和交叉引用；在上述宏包之后加载。
\usepackage[
    hidelinks
]{hyperref}
