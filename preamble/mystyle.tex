% 与 HTML <mark> 相同的纯黄底色；零内边距，不改变短语尺寸。
% \colorbox 把短语作为不可拆分的行内盒子，局部分组防止间距设置外泄。
\newcommand{\Highlight}[1]{%
  \begingroup
  \setlength{\fboxsep}{0pt}%
  \colorbox[HTML]{FFFF00}{#1}%
  \endgroup
}

% 代码卡片：浅灰标题栏、细灰边框、小圆角；代码设置仍由 \setminted 控制。
\definecolor{InputCodeTitle}{HTML}{F5F5F7}
\definecolor{InputCodeFrame}{HTML}{D1D1D6}
\definecolor{InputCodeText}{HTML}{1D1D1F}
\definecolor{InputCodeMuted}{HTML}{6E6E73}
% 只用于标题图标；不改变文件名、代码正文或行内代码字体。
\newfontfamily\InputCodeIconFont{Maple Mono NF CN}
\ExplSyntaxOn
\str_new:N \l__inputcode_area_str
\str_new:N \l__inputcode_base_str
\str_new:N \l__inputcode_ext_str
\str_new:N \l__inputcode_title_str

% Nerd Fonts 官方 glyphnames.json 中的语言图标；其余语言使用 fa-file_code。
% https://github.com/ryanoasis/nerd-fonts/blob/master/glyphnames.json
\cs_new_protected:Npn \__inputcode_icon:n #1
  {
    \group_begin:
    \InputCodeIconFont\mdseries\upshape\color{InputCodeMuted}
    \str_case:nnF {#1}
      {
        {cpp}{\symbol{"E61D}} % custom-cpp
        {c++}{\symbol{"E61D}}
        {c}{\symbol{"E61E}}   % custom-c
        {python}{\symbol{"E606}} % seti-python
        {py}{\symbol{"E606}}
        {bash}{\symbol{"EBCA}} % cod-terminal_bash
        {sh}{\symbol{"EBCA}}
        {shell}{\symbol{"EBCA}}
        {make}{\symbol{"E673}} % seti-makefile
        {makefile}{\symbol{"E673}}
      }
      {\symbol{"F1C9}} % fa-file_code
    \group_end:
  }

\cs_new_protected:Npn \__inputcode_filename:n #1
  {
    \file_parse_full_name:nNNN {#1}
      \l__inputcode_area_str \l__inputcode_base_str \l__inputcode_ext_str
    \str_set:Nx \l__inputcode_title_str
      {\str_use:N \l__inputcode_base_str \str_use:N \l__inputcode_ext_str}
  }

\cs_new_protected:Npn \__inputcode_print_title:n #1
  {
    \str_map_inline:nn {#1}
      {
        \str_if_eq:nnTF {##1} { ~ }
          {\hskip\fontdimen2\font\relax}
          {##1\penalty200\relax}
      }
  }

\cs_new_protected:Npn \__inputcode_title:nn #1#2
  {
    \xeCJKsetup{CJKecglue={}}
    \raggedright
    % 图标独占 4 mm；多行文件名的续行与首行对齐。
    \hangindent=4mm\hangafter=1
    \noindent\makebox[4mm][l]{\__inputcode_icon:n {#2}}
    \__inputcode_print_title:n {#1}\par
  }

\cs_new_protected:Npn \__inputcode_box:nnn #1#2#3
  {
    \par
    \begin{tcolorbox}[
      enhanced,
      breakable,
      title={\__inputcode_title:nn {#1}{#2}},
      colback=white,
      colbacktitle=InputCodeTitle,
      colframe=InputCodeFrame,
      coltitle=InputCodeText,
      boxrule=0.4pt,
      titlerule=0pt,
      arc=2mm,
      boxsep=0pt,
      left=2.5mm,right=2.5mm,top=2mm,bottom=2mm,
      lefttitle=2.5mm,righttitle=2.5mm,toptitle=1.5mm,bottomtitle=1.5mm,
      fonttitle=\normalfont,
      % 各版可设置此样式；未设置时保留接口默认外观。
      input-code-title/.try,
      % 前后距由各版设定；未设置时沿用 tcolorbox 的默认前后距。
      book-box-spacing/.try
    ]
    \xeCJKsetup{CJKecglue={}}
    % 行号向卡片左边框外悬挂；零宽度编号盒不挤占代码区，也不影响折行。
    % numbersep 从代码左缘量起：左内距 2.5 mm + 边框 0.4 pt + 框外 1 mm。
    \renewcommand{\theFancyVerbLine}{{%
      \use:c {FV@FontFamily}\mdseries\upshape\fontsize{4}{10.5}\selectfont
      \color[HTML]{808080}\arabic{FancyVerbLine}}}
    % 上下留白由卡片的 top/bottom 管理，不叠加 FancyVerb 列表间距。
    \inputminted[mathescape=false,texcomments=false,autogobble=false,
      linenos=true,firstnumber=1,stepnumber=1,numberblanklines=true,
      numbersep=\dimexpr 3.5mm+0.4pt\relax,
      listparameters={\setlength{\topsep}{0pt}\setlength{\partopsep}{0pt}},
      vspace=0pt]{#2}{#3}
    \end{tcolorbox}
  }

\NewDocumentCommand{\InputCode}{o m m}
  {
    \IfNoValueTF{#1}
      {
        \__inputcode_filename:n {#3}
        \exp_args:NV \__inputcode_box:nnn \l__inputcode_title_str {#2} {#3}
      }
      {\__inputcode_box:nnn {#1} {#2} {#3}}
  }
\ExplSyntaxOff

% 可跨页的引用块：浅灰背景、左侧深灰竖线，无其余边框。
\definecolor{QuoteBlockBackground}{HTML}{F7F7F7}
\definecolor{QuoteBlockRule}{HTML}{777777}
\newtcolorbox{quoteblock}{
  enhanced,
  breakable,
  % 保留所在版式的段间距与首行缩进，避免默认 parbox 将其清零。
  parbox=false,
  colback=QuoteBlockBackground,
  boxrule=0pt,
  borderline west={2pt}{0pt}{QuoteBlockRule},
  arc=0pt,
  boxsep=0pt,
  % 左竖线外侧至正文约一个中文字宽，包含竖线自身的厚度。
  left=\ccwd,right=2mm,top=2mm,bottom=2mm,
  % 前后距与代码卡片相同，由各版设定；未设置时沿用 tcolorbox 的默认前后距。
  book-box-spacing/.try
}

% Markdown 水平分隔线：按当前容器的 \linewidth 排版，不改变分页。
% 横线按正常行距排在上一行之后，下方另加 0.4 倍行距，补足下一行汉字的字面高度；
% 横线与上下两行文字的字面因此各隔约一行（用户确认）。
\definecolor{SeparatorRule}{HTML}{A0A0A0}
\NewDocumentCommand{\Separator}{}{%
  \par
  \noindent{\color{SeparatorRule}\rule{\linewidth}{0.4pt}}\par
  \addvspace{0.4\baselineskip}%
}
