\newcommand{\PortraitTOCEntry}[4]{%
  \par
  \ifnum\PortraitTOCPreviousLevel=1 \penalty10000
  \else\ifnum\PortraitTOCPreviousLevel=2
    \ifnum#1=3 \penalty10000\else\penalty0\fi
  \else\penalty0\fi\fi
  \begingroup
  \@tempdima=0pt
  \ifnum\PortraitTOCPreviousLevel=1 \advance\@tempdima by5pt\fi
  \ifnum#1=1 \advance\@tempdima by12pt\fi
  \vskip\@tempdima
  \nointerlineskip
  \vbox{%
    \hsize=\linewidth
    \normalfont\PortraitTOCSerif\fontsize{10}{15}\selectfont
    \ifnum#1=1 \PortraitTOCLatin\PortraitTOCCJK\fontsize{12}{15}\selectfont\fi
    \leftskip=#2\relax \rightskip=30pt
    \parfillskip=-\rightskip \parindent=0pt \parskip=0pt
    \interlinepenalty=10000
    \noindent\strut#3\nobreak
    \ifnum#1=1 \hfill
    % 9pt 是实际点中心间距；两端各留 3pt，空间不足一格时不绘点。
    \else{\normalfont\fontsize{10}{15}\selectfont\color{PortraitTOCDots}%
      \kern3pt\leaders\hbox to9pt{\hss.\hss}\hfill\kern3pt}\fi
    \nobreak\hb@xt@20pt{\hfil\normalfont\PortraitTOCSerif
      \ifnum#1=1 \fontsize{11}{15}\else\fontsize{10}{15}\fi\selectfont#4}\strut\par
  }%
  \endgroup
  \global\PortraitTOCPreviousLevel=#1\relax}
