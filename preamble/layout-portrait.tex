% 展示公式局部取消段距。段落后另起展示公式时，抵消 TeX 为进入
% 数学模式而产生的空行；公式后源码空行不再产生额外的 7bp 段距。
\makeatletter
\newcommand\PortraitDisplayBefore{%
  \ifvmode\ifdim\pagetotal>0pt\vskip-\baselineskip\fi\fi
  \begingroup\setlength{\parskip}{0pt}}
\newcommand\PortraitDisplayAfter{%
  \endgroup\par\vskip-\parskip\ignorespaces}
\makeatother
\BeforeBeginEnvironment{equation}{\PortraitDisplayBefore}
\AfterEndEnvironment{equation}{\PortraitDisplayAfter}
\BeforeBeginEnvironment{equation*}{\PortraitDisplayBefore}
\AfterEndEnvironment{equation*}{\PortraitDisplayAfter}
\BeforeBeginEnvironment{align}{\PortraitDisplayBefore}
\AfterEndEnvironment{align}{\PortraitDisplayAfter}
\BeforeBeginEnvironment{align*}{\PortraitDisplayBefore}
\AfterEndEnvironment{align*}{\PortraitDisplayAfter}
\BeforeBeginEnvironment{gather}{\PortraitDisplayBefore}
\AfterEndEnvironment{gather}{\PortraitDisplayAfter}
\BeforeBeginEnvironment{gather*}{\PortraitDisplayBefore}
\AfterEndEnvironment{gather*}{\PortraitDisplayAfter}
