\newcommand{\CodeTOCEntry}[4]{%
  \par
  \ifnum\CodeTOCPreviousLevel=1 \penalty10000
  \else\ifnum\CodeTOCPreviousLevel=2
    \ifnum#1=3 \penalty10000\else\penalty0\fi
  \else\penalty0\fi\fi
  \begingroup
  \@tempdima=0pt
  \ifnum\CodeTOCPreviousLevel=1 \advance\@tempdima by5pt\fi
  \ifnum#1=1 \advance\@tempdima by12pt\fi
  \vskip\@tempdima
  \nointerlineskip
  \vbox{%
    \hsize=\linewidth
    \normalfont\fontsize{10}{15}\selectfont
    \ifnum#1=1 \sffamily\bfseries\fontsize{12}{15}\selectfont
    \else\ifnum#1=2 \fontsize{10.5}{15}\selectfont\fi\fi
    \leftskip=#2\relax \rightskip=30pt
    \parfillskip=-\rightskip \parindent=0pt \parskip=0pt
    \interlinepenalty=10000
    \noindent\strut#3\nobreak
    \ifnum#1=1 \hfill
    % 9pt 是实际点中心间距；两端各留 3pt，空间不足一格时不绘点。
    \else{\normalfont\fontsize{10}{15}\selectfont\color{CodeTOCDots}%
      \kern3pt\leaders\hbox to9pt{\hss.\hss}\hfill\kern3pt}\fi
    \nobreak\hb@xt@20pt{\hfil\normalfont
      \ifnum#1=1 \fontsize{11}{15}\else\fontsize{10}{15}\fi\selectfont#4}\strut\par
  }%
  \endgroup
  \global\CodeTOCPreviousLevel=#1\relax}

% 页眉显示页顶所在的一级标题（用户确认，2026-10-14）：
% 本页没有新章，或新章不在左栏顶端时，取页首延续的章（TopMark）；
% 本页第一个新章恰好位于左栏顶端时，取该章。章节连续排版，
% 一页可含多章，不能沿用竖版取本页最后标记的 \leftmark。
% 位置由标题文字内的 \pdfsavepos 记录，需两轮编译；run.sh 至少两轮。
\makeatletter
\NewMarkClass{code-chapter}
\newcounter{CodeChapter}
\AddToHook{cmd/section/before}{\stepcounter{CodeChapter}}
% 置于标题文字内部，保证位置与标题同在一栏，不会因断页与标题分离。
\newcommand{\CodeChapterAnchor}[1]{%
  \leavevmode\pdfsavepos
  \RecordProperties{CodeChapter@\the\value{CodeChapter}}{xpos,ypos}#1}
\newdimen\CodeChapterTopLimit
\newcommand{\CodeHeaderChapter}{%
  \begingroup
  \edef\CodeTop{\TopMark{code-chapter}}%
  \edef\CodeFirst{\FirstMark{code-chapter}}%
  \let\CodeShown\CodeTop
  \ifx\CodeTop\CodeFirst\else
    \ifx\CodeTop\@empty\let\CodeShown\CodeFirst\fi
    % 左栏顶端标题的基线距版心顶约一个 topskip；留 20pt 容差，
    % 上方哪怕只有一行正文，标题基线也会再低约 30pt。
    \CodeChapterTopLimit=\dimexpr\paperheight-1in-\voffset-\topmargin
      -\headheight-\headsep-\topskip-20pt\relax
    \ifdim\RefProperty{CodeChapter@\CodeFirst}{xpos}sp<.5\paperwidth
      \ifdim\RefProperty{CodeChapter@\CodeFirst}{ypos}sp>\CodeChapterTopLimit
        \let\CodeShown\CodeFirst
      \fi
    \fi
  \fi
  \ifx\CodeShown\@empty\else
    \csname CodeChapterName@\CodeShown\endcsname
  \fi
  \endgroup}
\makeatother
